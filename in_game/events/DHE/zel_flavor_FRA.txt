	type = country_event

	desc = zel_flavor_fra.1.desc
	outcome = neutral

	fire_only_once = yes

		tag = FRA
		from = 1337.1.1
		to = 1340.1.1
		monthly_chance = 100
	}

	illustration_tags = {
		10 = angry
		10 = interior
	}

	immediate = {
		add_country_modifier = {
			modifier = zel_france_nerf
			years = 40
			mode = replace
		}

		add_country_modifier = {
			modifier = zel_france_trib
			years = -1
			mode = replace
		}
	}

	option = {
		name = zel_flavor_fra.1.a

		show_as_tooltip = {
			add_country_modifier = {
				modifier = zel_france_nerf
				years = 40
				mode = replace
			}
	
			add_country_modifier = {
				modifier = zel_france_trib
				years = -1
				mode = replace
			}
		}
		trigger_event_non_silently = {
			id = zel_flavor_fra.2
		}
	}
}
